cmake_minimum_required(VERSION 3.21)
project(ELIT21ShopManager VERSION 9.0.0 LANGUAGES CXX)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(CompilerOptions)
include(CompilerWarnings)
include(Sanitizers)
include(StaticAnalysis)
include(Dependencies)
include(InstallLayout)

option(ELIT21_BUILD_TESTS "Build all tests" ON)
option(ELIT21_BUILD_TOOLS "Build diagnostic tools" ON)
option(ELIT21_BUILD_EXTENDED_APPS "Build extended service applications" ON)
set(ELIT21_UNITY_BATCH_SIZE "16" CACHE STRING "Number of C++ source files per Unity translation unit")
if(NOT ELIT21_UNITY_BATCH_SIZE MATCHES "^[1-9][0-9]*$")
    message(FATAL_ERROR "ELIT21_UNITY_BATCH_SIZE must be a positive integer")
endif()

file(GLOB_RECURSE ELIT21_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
add_library(elit21_core STATIC ${ELIT21_SOURCES})
target_include_directories(elit21_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(elit21_core PUBLIC PkgConfig::LIBCURL OpenSSL::Crypto SQLite::SQLite3 PkgConfig::JSONC)
target_compile_definitions(elit21_core PUBLIC ELIT21_VERSION="${PROJECT_VERSION}")
set_target_properties(elit21_core PROPERTIES
    UNITY_BUILD ON
    UNITY_BUILD_BATCH_SIZE ${ELIT21_UNITY_BATCH_SIZE})
message(STATUS "ELIT21 Unity batch size: ${ELIT21_UNITY_BATCH_SIZE}")
if(WIN32)
    target_link_libraries(elit21_core PUBLIC ws2_32 crypt32)
    target_compile_definitions(elit21_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()

function(elit21_add_entrypoint source target)
    add_executable(${target} "${source}")
    target_link_libraries(${target} PRIVATE elit21_core)
endfunction()

elit21_add_entrypoint("apps/manager/main.cpp" elit21_shop_manager)
elit21_add_entrypoint("apps/diagnostics/main.cpp" elit21_api_diagnostics)

if(ELIT21_BUILD_EXTENDED_APPS)
    file(GLOB ELIT21_APP_MAINS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/apps/*/main.cpp")
    foreach(app_main IN LISTS ELIT21_APP_MAINS)
        get_filename_component(app_dir "${app_main}" DIRECTORY)
        get_filename_component(app_name "${app_dir}" NAME)
        if(NOT app_name STREQUAL "manager" AND NOT app_name STREQUAL "diagnostics")
            elit21_add_entrypoint("${app_main}" "elit21_${app_name}")
        endif()
    endforeach()
endif()

if(ELIT21_BUILD_TOOLS)
    file(GLOB ELIT21_TOOL_MAINS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tools/*/main.cpp")
    foreach(tool_main IN LISTS ELIT21_TOOL_MAINS)
        get_filename_component(tool_dir "${tool_main}" DIRECTORY)
        get_filename_component(tool_name "${tool_dir}" NAME)
        elit21_add_entrypoint("${tool_main}" "elit21_tool_${tool_name}")
    endforeach()
endif()

if(ELIT21_BUILD_TESTS)
    enable_testing()
    add_executable(elit21_tests tests/unit_tests.cpp)
    target_link_libraries(elit21_tests PRIVATE elit21_core)
    target_compile_definitions(elit21_tests PRIVATE ELIT21_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
    add_test(NAME elit21_tests COMMAND elit21_tests)
    add_subdirectory(tests)
endif()

install(TARGETS elit21_shop_manager elit21_api_diagnostics RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY config resources migrations docs DESTINATION ${CMAKE_INSTALL_DATADIR}/elit21)
include(Packaging)
